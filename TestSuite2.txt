LoadTestDataExcel
    [Arguments]    ${sTestCaseNo}
    Set Suite Variable    ${sFileName}    ${TestDataFileName}
    Set Suite Variable    ${sSheetName}    ${TestDataSheetName}
    Set Suite Variable    ${bDataFound}    ${TC_False}
    log    ${sFileName}
    Open Excel    ${sFileName}
    #Get the total rows in the Sheet
    ${iTotalRows}    ExcelLibrary.Get Row Count    ${sSheetName}
    : FOR    ${iRowNo}    IN RANGE    1    ${iTotalRows}
    \    ${TC_Num}    Read Cell Data By Coordinates    ${sSheetName}    0    ${iRowNo}
    \    log    ${sTestCaseNo}
    \    Run Keyword If    '${sTestCaseNo}'=='${TC_Num}'    Set Suite Variable    ${bDataFound}    ${TC_True}
    \    ${sSearchedData}    Run Keyword If    '${sTestCaseNo}'=='${TC_Num}'    Get Row Values    ${sSheetName}    ${iRowNo}
    \    ${arrTestData}    Run Keyword If    '${sTestCaseNo}'=='${TC_Num}'    Convert To String    ${sSearchedData}
    \    ${sKey}    Run Keyword If    '${sTestCaseNo}'=='${TC_Num}'    GetSubstringByDelimiter    ${arrTestData}    (
    \    ...    2
    \    ${sKey}    Run Keyword If    '${sTestCaseNo}'=='${TC_Num}'    GetSubstringByDelimiter    ${sKey}    '
    \    ...    3
    \    ${sValue}    Run Keyword If    '${sTestCaseNo}'=='${TC_Num}'    GetSubstringByDelimiter    ${arrTestData}    (
    \    ...    3
    \    ${sValue}    Run Keyword If    '${sTestCaseNo}'=='${TC_Num}'    GetSubstringByDelimiter    ${sValue}    '
    \    ...    3
    \    Run Keyword If    '${sTestCaseNo}'=='${TC_Num}'    Set To Dictionary    ${dTestData}    ${sKey}    ${sValue}
    \    Exit For Loop If    '${sTestCaseNo}'<>'${TC_Num}' and '${bDataFound}'== '${TC_True}'
    Set Suite Variable    &{dTestData}

LoadTestDataMySQL
    [Arguments]    ${sTestCaseNo}
    Connect To Database Using Custom Params    pymysql    database='${MySQLDB}', user='${MySQLUser}', password='${MySQLPassword}', host='${MySQLHost}', port=${MySQLPort}
    ${sqlQuery}    catenate    SELECT TDesc,TDescVal FROM    ${MySQLTableName}    WHERE TCID=    "${sTestCaseNo}"
    ${ResultSet}    Query    ${sqlQuery}
    Log Many    ${ResultSet}
    ${TC_COUNTER}    Get Length    ${ResultSet}
    : FOR    ${TC_LOOPCOUNER}    IN RANGE    0    ${TC_COUNTER}
    \    Set To Dictionary    ${dTestData}    ${ResultSet[${TC_LOOPCOUNER}][0]}    ${ResultSet[${TC_LOOPCOUNER}][1]}
    Return    ${dTestData}
